# ==================== 3. 이분 탐색: 최초 참 ====================
def first_true(lo, hi, ok):
    """ok(x)가 False -> True로 단조 변화하고 ok(hi)가 True일 때 첫 True."""
    while lo < hi:
        mid = (lo + hi) // 2
        if ok(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo


def kth_multiplication_table(n, k):
    """n*n 곱셈표의 k번째 원소 (1<=k<=n*n), O(n log n)."""
    def count_leq(x):
        return sum(min(n, x // i) for i in range(1, n + 1))
    # count(mid) == k여도 중단 금지! 답은 count(x)>=k인 최소 x.
    return first_true(1, n * n, lambda x: count_leq(x) >= k)


# ==================== 4. DP: 편집 거리 ====================
def edit_distance(a, b):
    """삽입/삭제/교체 비용 각 1, O(len(a)*len(b)) 시간/공간."""
    n, m = len(a), len(b)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dp[i][0] = i             # 빈 문자열로 만드는 삭제 비용
    for j in range(m + 1):
        dp[0][j] = j             # 빈 문자열부터 삽입 비용
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            dp[i][j] = min(dp[i-1][j] + 1,
                           dp[i][j-1] + 1,
                           dp[i-1][j-1] + (a[i-1] != b[j-1]))
    return dp[n][m]

# DP 체크: dp의 의미 -> 초기값 -> 점화식 -> 순회순서 -> 최종 위치
# 최대값 DP의 불가능한 상태는 -INF, 경우의 수 DP는 MOD 적용 확인
# 1차원 DP 압축 시 이전 행/현재 행 값이 섞이지 않는지 확인

# ==================== 5. 좌표기하 ====================
def sub(a, b):
    return (a[0] - b[0], a[1] - b[1])


def dot(u, v):
    return u[0] * v[0] + u[1] * v[1]


def cross(u, v):
    return u[0] * v[1] - u[1] * v[0]


def dist2(a, b):
    return dot(sub(a, b), sub(a, b))  # 거리 비교 시 sqrt 생략


def area2(a, b, p):
    return abs(cross(sub(b, a), sub(p, a)))  # 삼각형 넓이의 2배


def perpendicular_foot(a, b, p):
    """직선 AB 위 점 P의 수선의 발 H와 비율 t; A!=B 필요."""
    v = sub(b, a)
    denom = dot(v, v)
    if denom == 0:
        raise ValueError('A와 B가 같아 직선이 정의되지 않음')
    t = dot(sub(p, a), v) / denom
    h = (a[0] + t * v[0], a[1] + t * v[1])
    return h, t             # 0<=t<=1이면 발이 선분 AB 내부/끝


def point_line_dist(a, b, p):
    """점 P와 무한 직선 AB의 거리 (A!=B)."""
    return area2(a, b, p) / math.sqrt(dist2(a, b))


def reflected_point(a, b, p):
    """점 P를 무한 직선 AB에 대칭시킨 점."""
    h, _ = perpendicular_foot(a, b, p)
    return (2 * h[0] - p[0], 2 * h[1] - p[1])


def intersection_x(a, b, c):
    """직선 AB와 x=c의 교점, 평행이면 None (일치도 None)."""
    dx = b[0] - a[0]
    if dx == 0:
        return None
    t = (c - a[0]) / dx
    return (c, a[1] + t * (b[1] - a[1]))


def intersection_y(a, b, c):
    """직선 AB와 y=c의 교점, 평행이면 None (일치도 None)."""
    dy = b[1] - a[1]
    if dy == 0:
        return None
    t = (c - a[1]) / dy
    return (a[0] + t * (b[0] - a[0]), c)
